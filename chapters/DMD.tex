\section{Frobeniusnorm}\label{chap:Frobeniusnorm}

\begin{definition}[Frobeniusnorm]
	Sei $X\in\CC^{p\times n}$ eine Matrix und $x_{ij}$ mit $i\in \{1,\dots,p\},$ $j\in \{1,\dots,n\}$ ihre Einträge. Die \emph{Frobeniusnorm} von $X$ ist definiert durch
	\[
		||X||_F \coloneqq \sqrt{\sum_{i=1}^{p}\sum_{j=1}^{n}|x_{ij}|^2}\,.
	\]
\end{definition}
Die Frobeniusnorm können wir aber auch über die Spur der Matrix beschreiben.
\begin{lemma}\label{lem:FrobTrace}
	Sei $X\in\CC^{p\times n}$ eine Matrix. Dann gilt
	\[
		||X||_F = \sqrt{\tr\left( X^HX\right)}\,.
	\]
\end{lemma}

\begin{proof}
	Sei $x_i$ der $i$-te Spaltenvektor von $X$ und $\overline{x}_{ij}$ das komplex Konjugierte von $x_{ij}$. Dann ist
	\begin{align*}
		X^HX & =
		\begin{pmatrix}
			x_1 & \cdots & x_n 
		\end{pmatrix}^H
		\begin{pmatrix}
			x_1 & \cdots & x_n
		\end{pmatrix} \\
		     & = \begin{pmatrix}
			         \displaystyle\sum_{j=1}^{p}\langle x^{H}_{1}, x_{1}\rangle &        & *                                             \\
			                                                      & \ddots &                                               \\
			         *                                            &        & \displaystyle\sum_{j=1}^{p} \langle x^{H}_{p}, x_{p}\rangle
		         \end{pmatrix}
		= \begin{pmatrix}
			  \displaystyle\sum_{j=1}^{p}\overline{x}_{1j} x_{1j} &        & *                                                    \\
			                                                      & \ddots &                                                      \\
			  *                                                   &        & \displaystyle\sum_{j=1}^{p} \overline{x}_{nj} x_{nj}
		  \end{pmatrix}\,.
	\end{align*}

	Mit \(\overline{x}x = |x|^2\) erhalten wir
	\begin{align}
		\tr \left(X^HX\right) = \sum_{i=1}^{n}\sum_{j=1}^{p} \overline{x}_{ij} x_{ij} = \sum_{i=1}^{n}\sum_{j=1}^{p} |x_{ij}|^2 = ||X||_F^2 \,. \label{eq:traceFrob}
	\end{align}
\end{proof}


Die Frobeniusnorm wird im weiteren Verlauf aufgrund ihrer Invarianz bezüglich unitärer Transformation verwendet, welche wir durch folgendes Lemma zeigen.

\begin{lemma}
	Sei $X\in\CC^{p\times n}$ eine Matrix und \(U\in\CC^{p\times p}\) und \(V\in\CC^{n\times n}\) unitäre Matrizen. Dann gilt
	\begin{enumerate}
		\item \(||X||_F =||X^H||_F\),
		\item \(||UX||_F =||X||_F\),
		\item \(||XV||_F =||X||_F\).
	\end{enumerate}
\end{lemma}
\begin{proof}\textcolor{white}{Wackelpudding} %\emph{Beweis.}
	\begin{enumerate}
		\item Aus Lemma \ref{lem:FrobTrace} folgt
		      \[
			      ||X||_F^2 = \tr \left(X^HX\right) = \sum_{i=1}^{n}\sum_{j=1}^{p} \overline{x}_{ij} x_{ij} = \sum_{j=1}^{p}\sum_{i=1}^{n} \overline{x}_{ij} x_{ij} = \tr\left(XX^H\right) = ||X^H||_F^2 \,.
		      \]
		\item Es ist
		      \[
			      ||UX||_F^2 = \tr \left(X^HU^HUX\right) =\tr \left(X^H\mathbb{1}X\right) = \tr \left(X^HX\right) = ||X||_F^2\,.
		      \]
		\item Mit 1. und 2. folgt
		      \[
			      ||XV||_F = ||V^HX^H||_F = ||X^H||_F = ||X||_F\,. \qedhere
		      \]%\begin{flushright} $\square$ \end{flushright}
	\end{enumerate}
\end{proof}

\section{Singulärwertzerlegung}\label{chap:SWZ}

\begin{theorem}[Singulärwertzerlegung {\cite[228]{Numerik1}}] \label{theo:svd}
	Sei \(X\in \mathbb{R}^{p\times n} \) eine Matrix und ${r\coloneqq \rang X}$ mit $r\leq \min(p,n)$.
	Dann existieren orthogonale Matrizen \(U\in\RR^{p\times p}\) und \(V\in\RR^{n\times n}\) und eine Diagonalmatrix \(\hat{\Sigma}=\diag(\sigma_1,\dots,\sigma_r)\in\mathbb{R}^{r\times r}\) mit $\sigma_1 \geq \sigma_2 \geq \dots \geq \sigma_r > 0$ so, dass 
	\[
		X = U\Sigma V^\top \quad\text{mit}\quad \Sigma = \begin{pmatrix}
			\hat{\Sigma} & 0 \\
			0            & 0
		\end{pmatrix} \in \mathbb{R}^{p\times n}
	\]
	gilt.
	Wir nennen die Diagonalelemente von $\Sigma$ Singulärwerte.
\end{theorem}


Der Beweis des Satzes \ref{theo:svd} folgt etwas später, da dafür noch etwas Vorbereitung benötigt wird.
Wichtig zu erwähnen ist noch, dass unter den in Satz \ref{theo:svd} genannten Voraussetzungen, $\hat{\Sigma}$ durch $X$ eindeutig bestimmt ist \cite[229]{Numerik1}.
Die beiden orthogonalen Matrizen $U$ und $V$ hingegen sind nicht eindeutig.

\begin{lemma}\label{lem:EigenwerteSingulärwerte}
	Seien \(X\in \mathbb{R}^{p\times n}\) und $\sigma_i$ die Singulärwerte von $X$ nach den Voraussetzungen aus Satz \ref{theo:svd}. 
	Seien weiter $\lambda_i$ die Eigenwerte von $X^\top X$. 
	Dann gilt 
	\begin{align}
		\sigma_i=\sqrt{\lambda_i}\,.\label{eq:SWzuEW}
	\end{align}
		
\end{lemma}

\begin{proof}
	Mit $U$ und $V$ aus Satz \ref{theo:svd} ist durch 
	\begin{align*}
		X^\top X &= \left(U\Sigma V^\top\right)^\top\left(U\Sigma V^\top\right)\\
				 &= V\Sigma^\top U^\top U\Sigma V^\top \\
				 &= V\Sigma^\top \Sigma V^\top
	\end{align*}
	ist $X^\top X$ ähnlich zu $\Sigma^\top \Sigma$. 
	Also haben $X^\top X$ und $\Sigma^\top \Sigma$ auch die gleichen Eigenwerte.
	Damit sind auch die Wurzeln der Eigenwerte von $X^\top X$ gleich der Eigenwerte beziehungsweise Singulärwerte von $\Sigma$.
\end{proof}

\begin{lemma} \label{lem:InvarinatBzglTansp}
	Die aus der Singulärwertzerlegung in \ref{theo:svd} entstehenden Singulärwerte sind invariant bezüglich Transposition vom $X$.
\end{lemma}

\begin{proof}
	Seien die Voraussetzungen aus Satz \ref{theo:svd} gegeben, dann gilt
	\begin{align*}
		X^\top = \left(U\Sigma V^\top\right)^\top = V\Sigma^\top U^\top\,.
	\end{align*}
	Da $\Sigma$ nur auf der Hauptdiagonalen Einträge hat und sonst mit Nullen gefüllt ist, erhalten wir
	\[
		\diag \Sigma^\top = \{\sigma_1,\sigma_2,\dots,\sigma_m\} =\diag \Sigma\,.
	\]
\end{proof}

Den Beweis von Satz \ref{theo:svd} teilen wir auf. Wir betrachten dafür zum Einen die Singulärwertzerlegung für den Fall, dass $X$ vollen Rang besitzt und zum Anderen für den rangreduzierten Fall.
Für den vollen Rang benutzen wir das folgende Lemma.

\begin{lemma}[{\cite[191]{Numerik1}}] \label{lem:diagonal}
	Sei \(X\in\RR^{p\times p}\) eine symmetrische Matrix. Dann existiert eine orthogonale Matrix \(U\in\RR^{p\times p}\) und eine Diagonalmatrix $D\in\RR^{p\times p}$ so, dass
	\[
		U^\top XU = D
	\]
	gilt.
\end{lemma}

Für den rangreduzierten Fall brauchen wir aber das folgende Lemma.
Dafür nehmen wir nach Lemma \ref{lem:InvarinatBzglTansp} o.B.d.A an, dass $n\leq p$ gilt.

\begin{lemma}[{\cite[227]{Numerik1}}] \label{lem:obereDreiecks}
	Sei \(X\in \mathbb{R}^{p\times n}\) eine Matrix mit \(\rang X = r < n \leq p\). Dann existieren orthogonale Matrizen \(U\in\RR^{p\times p}\) und \(V\in\RR^{n\times n}\) sowie eine reguläre obere Dreiecksmatrix $\hat{R}\in\RR^{r\times r}$ so, dass
	\[
		U^\top X V=R \quad\text{mit}\quad R = \begin{pmatrix}
			\hat{R} & 0 \\
			0       & 0
		\end{pmatrix} \in \RR^{p\times n}
	\]
	gilt.
\end{lemma}

Für den Beweis gehen wir mit Lemma \ref{lem:InvarinatBzglTansp} o.B.d.A davon aus, dass $n<p$ gilt.

\begin{proof}[Beweis von Satz \ref{theo:svd}]
	Zunächst betrachten wir den Fall des vollen Ranges von $X$, also \\ $\rang X = r = n$.
	Bei vollem Rang ist \(X^\top X\in\mathbb{R}^{n\times n}\) sowohl symmetrisch als auch positiv definit. Die Eigenwerte \(\lambda_i\) von \(X^\top X\) sortieren wir nach ihrer Größe, also
	\[
		\lambda_1 \geq \lambda_2 \geq \dots \geq \lambda_n > 0\,.
	\]
	Mit Lemma \ref{lem:diagonal} finden wir ein orthogonales \(V \in \RR^{n\times n}\) so, dass
	\[
		V^\top X^\top XV=D=\diag (\lambda_1, \dots , \lambda_n) \in \mathbb{R}^{n\times n}\,.
	\]
	Weiter definieren wir
	\[
		\hat{\Sigma}\coloneqq \diag (\sigma_1, \dots, \sigma_n)
	\]
	mit \(\sigma_i \coloneqq \sqrt{\lambda_i}\in(0,\infty),\text{ für } i=1,\dots, n\), und setzen
	\[
		\hat{U}\coloneqq XV\hat{\Sigma}^{-1}\in\RR^{p\times n}\,.
	\]
	Jetzt kann man durch Umformung von
	\begin{align*}
		\hat{U}^\top\hat{U} & = \hat{\Sigma}^{-\top}V^\top X^\top XV\hat{\Sigma}^{-1} \\
		                    & =  \hat{\Sigma}^{-1}D\hat{\Sigma}^{-1}                  \\
		                    & =  \hat{\Sigma}^{-1}\hat{\Sigma}^{-1}D                  \\
		                    & =  D^{-1}D                                              \\
		                    & = \mathbb{1}
	\end{align*}
	die Orthonormalität der Spalten von \(\hat{U}\) sehen.

	Da $\hat{U}$ in $\RR^{p\times n}$ liegt und aus orthonormalen Spaltenvektoren besteht, finden wir ein \(Z\in\RR^{p\times (p-n)}\), welches $\hat{U}$ so um $p-n$ orthonormale Spaltenvektoren ergänzt, dass
	\[
		U = (\hat{U}\ Z)\in\RR^{p\times p}
	\]
	eine orthogonale Matrix ist. Es gilt also
	\begin{align*}
		Z^\top XV\hat{\Sigma}^{-1} & =Z^\top\hat{U}               \\
		                           & =0 \in\RR^{(p-n)\times p}\,.
	\end{align*}
	Da \(\hat{\Sigma}\hat{\Sigma}^{-1}=\mathbb{1}\) ist, folgt auch direkt
	\[
		Z^\top XV=0\,.
	\]
	Zusammengetragen haben wir dann
	\begin{align*}
		U^\top XV & = \begin{pmatrix}
			              \hat{U}^\top \\
			              Z^\top
		              \end{pmatrix} XV  = \begin{pmatrix}
			                                  \hat{\Sigma}^{-1}V^\top X^TXV \\
			                                  Z^\top XV
		                                  \end{pmatrix} \\
		          & = \begin{pmatrix}
			              \hat{\Sigma}^{-1}D \\
			              0
		              \end{pmatrix}                                \\
		          & = \begin{pmatrix}
			              \hat{\Sigma} \\
			              0
		              \end{pmatrix}\,,
	\end{align*}
	womit wir den Fall des vollen Ranges bewiesen hätten.
	
	Seien $\lambda_i$ wie oben definiert wieder die Eigenwerte von $X^\top X$ und sei $X_r\in\RR^{p\times n}$ eine Matrix mit $\rang X_r=r<n$. Nach Lemma \ref{lem:obereDreiecks} gibt es zwei orthognonale Matrizen \(\tilde{U}\in\RR^{p\times p}\) und \(\tilde{V}\in\RR^{n\times n}\), sowie eine obere Dreiecksmatrix $\hat{R}\in\RR^{r\times r}$ so, dass
	\begin{align*}
		\tilde{U}^\top X_r \tilde{V}=R \text{ mit } R = \begin{pmatrix}
			                                                \hat{R} & 0 \\
			                                                0       & 0
		                                                \end{pmatrix} \in \RR^{p\times n}
	\end{align*}
	gilt. 
	Nach obigem Fall mit vollem Rang, wissen wir, dass es orthogonale Matrizen \(U_r\in\RR^{p\times p}\) und \(V_r\in\RR^{r\times r}\) mit
	\begin{align*}
		U_r^\top\begin{pmatrix}
			        \hat{R} \\
			        0
		        \end{pmatrix}V_r = \begin{pmatrix}
			                           \hat{\Sigma}_r \\
			                           0
		                           \end{pmatrix}\in\RR^{p\times r}
	\end{align*}
	gibt.
	Dabei ist $\hat{\Sigma}_r \coloneqq \diag (\sigma_1, \dots, \sigma_r)$ mit $\sigma_i = \sqrt{\lambda_i}$.
	Im Folgenden erweitern wir
	\begin{align*}
		\begin{pmatrix}
			\hat{\Sigma}_r \\
			0
		\end{pmatrix}\in\RR^{p\times r} \quad\text{zu}\quad\begin{pmatrix}
			                                                   \hat{\Sigma}_r & 0 \\
			                                                   0              & 0
		                                                   \end{pmatrix}=\Sigma \in \RR^{p\times n}
	\end{align*}
	und finden eine orthogonale Fortsetzung für $V_r$ zu
	\begin{align*}
		\overline{V}_r=\begin{pmatrix}
			          V_r & 0          \\
			          0   & \mathbb{1}
		          \end{pmatrix} \in \RR^{n\times n}\,.
	\end{align*}
	Wenn wir uns jetzt die orthogonalen Matrizen
	\begin{align*}
		U \coloneqq \tilde{U}U_r\in \RR^{p\times p}
		\quad\text{und}\quad
		V \coloneqq \tilde{V}\overline{V}_r\in \RR^{n\times n}
	\end{align*}
	definieren, erhalten wir durch
	\begin{align*}
		U^\top X_r V & = U_r^\top\tilde{U}X_r\tilde{V}\overline{V}_r \\
		             & =U_r^\top\begin{pmatrix}
			                        \hat{R} & 0 \\
			                        0       & 0
		                        \end{pmatrix}\overline{V}_r          \\
		             & =\begin{pmatrix}
			                \hat{\Sigma}_r & 0 \\
			                0              & 0
		                \end{pmatrix} \eqqcolon \Sigma
	\end{align*}
	eine erfolgreiche Zerlegung von $X_r$ durch orthogonale Matrizen.
\end{proof}

\section{Exact DMD}\label{chap:exactDMD}

In der Einleitung haben wir grob den DMD Algorithmus erklärt.
Wir gehen hier jetzt aber auf die \glqq Exact DMD\grqq\ Variante ein, da sie im Gegensatz zur ursprünglichen Variante keine einheitlichen Zeitabstände zwischen den Messungen erfordert.
Dieses Kapitel beruht auf dem Buch \glqq Data-Driven Science and Engineering\grqq\ \cite{SlaySlaySlay}.

Exact DMD erlaubt es uns Daten über mehrere Messungen die zeitlich unterschiedlich auseinanderliegen zusammenzutragen. Zu jeder Messung gehört dann ein Datenpaar $\{x(t_i),x(t_i')\}$ mit $i=1,\dots,n$ und $t'_i=t_i +\Delta t$.
Dabei ist $x$ die Messung zum Zeitpunkt $t_i$ beziehungsweise $t'_i$.
Fügen wir die Datenpaare in zwei Matrizen \(X,X'\in\mathbb{R}^{p\times n}\) ein, erhalten wir die Zustände des Systems zu \(n \) diskreten Zeitpunkten.
Wir erhalten also 
\begin{align*}
	X=\begin{pmatrix}
		|      & |      &       & |      \\
		x(t_1) & x(t_2) & \dots & x(t_n) \\
		|      & |      &       & |
	\end{pmatrix}
	\quad \text{und} \quad
	X'=\begin{pmatrix}
		|       & |       &       & |       \\
		x(t'_1) & x(t'_2) & \dots & x(t'_n) \\
		|       & |       &       & |
	\end{pmatrix}\,.
\end{align*}




Mit den Vorbereitungen abgeschlossen, gehen wir jetzt auf den Algorithmus ein.

Als Erstes bestimmen wir die Singulärwertzerlegung von $X$ also
\begin{align*}
	X=U\Sigma V^\top\,.
\end{align*}
Um den Rechenaufwand zu minimieren, finden wir eine geeignete Rangreduzierung.
Wir finden also ein $r\in\NN$ mit $r<n$ so, dass wir mit $U_r\in\RR^{p\times r}$, $\Sigma_r\in\RR^{r\times r}$ und $V_r\in\RR^{n\times r}$
\begin{align*}
	X\approx U_r\Sigma_rV_r^\top \eqqcolon X_r
\end{align*}
approximieren.
Um den diagonalisierbaren, linearen Operator $A$ zu finden, der 
\begin{align*}
	X'\approx AX
\end{align*}
löst benötigen wir die Pseudoinverse von $X$. Um eine eindeutige Inverse zu bekommen, benutzen wir die Moore-Penrose-Inverse aus dem folgenden Satz.
\begin{theorem}[Moore-Penrose-Inverse {\cite[230]{Numerik1}}] \label{theo:moorePenrose}
	Sei $X\in\RR^{p\times n}$ eine Matrix und $U, \hat{\Sigma}$ und $V$ die Matrizen aus der Singulärwertzerlegung in Satz \ref{theo:svd}. Dann existiert genau eine Moore-Penrose-Inverse der Form
	\begin{align*}
		X^\dag=V\begin{pmatrix}
			        \hat{\Sigma}^{-1} & 0 \\
			        0                 & 0
		        \end{pmatrix}U^\top \in\RR^{n\times p}\,.
	\end{align*}
\end{theorem}
Mit Satz \ref{theo:moorePenrose} können wir $A$ durch  
\begin{align*}
	A = X'V_r \Sigma_r^{-1} U_r^\top = X'X_r^\dagger
\end{align*}
bestimmen.
Das Ziel ist es, die Spektralzerlegung von $A$ zu finden, damit wir analysieren können wie sich das System mit der Zeit entwickelt.
Da $A$ aber eine $p\times p$ Matrix ist und $p$ schnell sehr groß wird, können wir den Rechenaufwand minimieren, indem wir die Spektralzerlegung von einem $A_r\in\RR^{r\times r}$ bestimmen, welches nur noch die für uns relevante Dimension $r$ hat.
Wir bestimmen $A_r$ durch
\begin{align}
	\begin{split}\label{eq:A_r}
	A_r&=U_r^\top A U_r \\
	 &= U_r^\top X'X^\dagger_r U_r \\
	 &= U_r^\top X'V_r\Sigma_r^{-1}\,.
	\end{split}
\end{align}

Sei $W\in\RR^{r\times r}$ die Matrix aus Eigenvektoren von $A_r$ und $\Lambda\in\RR^{r\times r}$ eine Diagonalmatrix dessen Einträge die Eigenwerte $\lambda_i$ der zugehörigen Eigenvektoren von $A_r$ sind, dann ist die Spektralzerlegung von $A_r$ genau
\begin{align}\label{eq:A_rW}
	A_rW=W\Lambda \,.
\end{align}
Sei $\Phi\in\RR^{p\times p}$ die Matrix, dessen Spalten die Eigenvektoren von $A$ sind. Dann können wir $\Phi$ durch die reduzierten Eigenvektoren $W$ mit 
\begin{align*}
	\Phi = X'V_r\Sigma_r^{-1} W
\end{align*}
rekonstruieren.

Durch
\begin{align*}
	A\Phi &= (X'V_r \Sigma_r^{-1} U_r^\top)(X'V_r\Sigma_r^{-1} W) \\
		  \overset{\eqref{eq:A_r}}&{=} X'V_r \Sigma_r^{-1} A_r W \\
		  \overset{\eqref{eq:A_rW}}&{=} X'V_r \Sigma_r^{-1} W\Lambda \\
		  &=\Phi \Lambda
\end{align*}
können wir sehen, dass die Spalten von $\Phi$ auch die Eigenvektoren zu den Eigenwerten von $A_r$ sind. Damit können wir also die Spektralzerlegung von $A$ mit der Spektralzerlegung von $A_r$ bestimmen.
